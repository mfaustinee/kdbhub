async function migrateValidations() {
  console.log("Fetching legacy validations from old project...");

  // Fetch rows from old table name (e.g. 'data_validations' or 'kdb_validations')
  const { data: oldRows, error: fetchErr } = await oldSupabase
    .from('kdb_validations')
    .select('*');

  if (fetchErr) {
    console.error("Error fetching from old project:", fetchErr);
    return;
  }

  console.log(`Found ${oldRows.length} records. Mapping columns to new schema...`);

  // Map old columns to new column names
  const mappedRows = oldRows.map(row => {
    const raw = row.raw_data || {};
    return {
      id: row.id,
      validation_period: row.validation_period || raw.validationPeriod || 'Monthly',
      date: row.date || raw.date || new Date().toISOString().split('T')[0],
      permit_no: row.permit_no || raw.permitNo || raw.permitNumber || '',
      dbo_name: row.dbo_name || raw.dboName || raw.clientName || '',
      premise_name: row.premise_name || raw.premiseName || '',
      location: row.location || raw.location || '',
      category: row.category || raw.category || '',
      contacts: row.contacts || raw.contacts || '',
      raw_data: raw
    };
  });

  // Upsert into new project
  const { data, error: insertErr } = await newSupabase
    .from('kdb_validations')
    .upsert(mappedRows, { onConflict: 'id' });

  if (insertErr) {
    console.error("Error inserting into new project:", insertErr);
  } else {
    console.log(`Successfully migrated ${mappedRows.length} validation records!`);
  }
}

migrateValidations();
